using System;
using System.Collections;
using System.IO;
using System.Collections.Generic;
using SDK_HD.KStruct.NativeRebar;
using Autodesk.Revit.DB;
using Autodesk.Revit.DB.Structure;

// API doubles exercise the production strict factory's flags/error paths, not native shape matching.
namespace Autodesk.Revit.DB
{
    public class ElementId { public long Value; public ElementId(long v){Value=v;} }
    public class Element { public ElementId Id; public string Name; }
    public class Document { public List<Element> Shapes=new List<Element>(); }
    public class XYZ {}
    public class Curve {}
    public class FilteredElementCollector : IEnumerable<Element>
    {
        readonly Document doc;
        public FilteredElementCollector(Document d){doc=d;}
        public FilteredElementCollector OfClass(Type t)=>this;
        public IEnumerator<Element> GetEnumerator()=>doc.Shapes.GetEnumerator();
        IEnumerator IEnumerable.GetEnumerator()=>GetEnumerator();
    }
}
namespace Autodesk.Revit.DB.Structure
{
    public enum RebarStyle { Standard,StirrupTie }
    public enum RebarHookOrientation { Left,Right }
    public class RebarShape : Element {}
    public class RebarBarType : Element {}
    public class RebarHookType : Element {}
    public class Rebar : Element
    {
        public static int Calls;
        public static bool Existing,NewShape;
        public static RebarHookType Start,End;
        public static Rebar Result;
        public static bool Throw;
        public ElementId GetShapeId()=>Id;
        public static Rebar CreateFromCurves(Document d,RebarStyle s,RebarBarType t,RebarHookType a,RebarHookType b,
            Element h,XYZ n,IList<Curve> c,RebarHookOrientation so,RebarHookOrientation eo,bool existing,bool create)
        {Calls++;Existing=existing;NewShape=create;Start=a;End=b;if(Throw)throw new InvalidOperationException("native shape mismatch");return Result;}
    }
}
static class Program
{
    static int checks;
    static void Check(bool ok,string message){checks++;if(!ok)throw new Exception(message);}
    static Rebar Create(Document d,RebarHookType start=null,RebarHookType end=null)=>ExistingProjectRebar.Create(d,
        RebarStyle.StirrupTie,new RebarBarType(),start,end,new Element(),new XYZ(),new List<Curve>{new Curve()},
        RebarHookOrientation.Left,RebarHookOrientation.Right,"Đai thử");
    static void Reject(Action a,string text){bool ok=false;try{a();}catch(InvalidOperationException e){ok=e.Message.Contains(text);}Check(ok,text);}
    static void Main()
    {
        var d=new Document();
        Reject(()=>Create(d),"chưa có Rebar Shape");Check(Rebar.Calls==0,"No creation without loaded shapes");
        d.Shapes.Add(new RebarShape {Id=new ElementId(10)});
        var hook=new RebarHookType {Name="Loaded project hook",Id=new ElementId(20)};
        Rebar.Result=new Rebar{Id=new ElementId(10)};
        Check(Create(d,hook,hook)==Rebar.Result,"Use matching pre-existing shape");
        Check(Rebar.Existing&&!Rebar.NewShape,"Native API must reuse shapes and prohibit shape creation");
        Check(Rebar.Start==hook&&Rebar.End==hook,"Pass exact selected project hook instances");
        Rebar.Result=null;Reject(()=>Create(d),"không tự sinh shape");
        Check(Rebar.Calls==2,"One attempt only; no fallback creation");
        Rebar.Result=new Rebar{Id=new ElementId(999)};Reject(()=>Create(d),"không tự sinh shape");
        Rebar.Throw=true;Reject(()=>Create(d,hook,hook),"native shape mismatch");
        Check(d.Shapes.Count==1,"Project shape inventory stays unchanged in strict factory test");
        var root=new DirectoryInfo(Directory.GetCurrentDirectory());
        while(root!=null&&!Directory.Exists(Path.Combine(root.FullName,".github")))root=root.Parent;
        Check(root!=null,"Locate repository for strict creation route checks");
        foreach(var file in new[]{"ColumnRebarPro/V3/V3RebarBuilder.cs","ColumnRebarPro/V3/V3RebarSetCreator.cs",
            "NativeRebar/BeamRebar/ReinforceStraightBeamsCommand.cs","NativeRebar/CurvedBeamRebar/ReinforceHorizontalCurvedBeamsCommand.cs",
            "NativeRebar/ReinforceLintelsCommand.cs"})
        {
            var source=File.ReadAllText(Path.Combine(root.FullName,file));
            foreach(var forbidden in new[]{"NativeRebarFactory.CreateFromCurves(","Rebar.CreateFromCurves(","Rebar.CreateFreeForm(","RebarShape.Create("})
                Check(!source.Contains(forbidden),file+" bypasses strict factory: "+forbidden);
            Check(source.Contains("ExistingProjectRebar.Create("),file+" must use existing-shape factory");
        }
        Console.WriteLine($"PASS: {checks} existing project shape policy checks (API doubles)");
    }
}
