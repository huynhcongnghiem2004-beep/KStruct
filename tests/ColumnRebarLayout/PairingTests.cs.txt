using System;
using System.Collections.Generic;
using System.Linq;
using ColumnRebarPro.V3;

static class PairingTests
{
    static int checks;
    static void Check(bool condition,string message){checks++;if(!condition)throw new Exception(message);}
    static void Near(double a,double b,string message)=>Check(Math.Abs(a-b)<1e-5,message);
    static void Reject(Action action,string message)
    {
        bool rejected=false;try{action();}catch(InvalidOperationException ex){rejected=ex.Message.Contains(message);}
        Check(rejected,"Expected rejection: "+message);
    }
    static V3ConnectionProfile Profile(double width=400,double depth=500,int cx=4,int cy=5,double angle=0,bool mirror=false)=>new V3ConnectionProfile
    {
        Label="Column",Width=width,Depth=depth,Cx=cx,Cy=cy,Diameter=20,TieDiameter=8,Cover=30,
        AxisX=new V3Point2(Math.Cos(angle),Math.Sin(angle)),
        AxisY=new V3Point2(-Math.Sin(angle)*(mirror?-1:1),Math.Cos(angle)*(mirror?-1:1))
    };
    static V3ColumnConnectionPlan Plan(V3ConnectionProfile a,V3ConnectionProfile b)
    {
        var plan=V3ColumnPairing.Create(a,b,2,6);
        int count=2*(a.Cx+a.Cy)-4;
        Check(plan.Bars.Count==count,"Every lower bar has a pair");
        Check(plan.Bars.Select(p=>p.UpperIndex).Distinct().Count()==count,"Every upper bar used once");
        foreach(var bar in plan.Bars)
        {
            Near((bar.End-bar.Upper).Length,(a.Diameter+b.Diameter)/2+2,"Correct paired diameters and clearance");
            Check(a.Contains(bar.End,a.Diameter)&&b.Contains(bar.End,a.Diameter),"Both cover envelopes");
            Near(bar.Height,bar.Shift*6,"Height uses actual displacement");
        }
        return plan;
    }
    internal static void Run()
    {
        Plan(Profile(),Profile());
        var shifted=Profile();shifted.Cover=40;shifted.TieDiameter=10;Plan(Profile(),shifted);
        var smaller=Profile(350,450);Plan(Profile(),smaller);
        var smallerSteel=Profile();smallerSteel.Diameter=16;Plan(Profile(),smallerSteel);
        var offset=Profile(350,450);offset.Center=new V3Point2(12,-8);Plan(Profile(),offset);
        foreach(double angle in new[]{0.0,Math.PI/6,Math.PI/2,Math.PI})
        foreach(bool mirror in new[]{false,true})
        {
            var a=Profile(angle:angle,mirror:mirror);var b=Profile(350,450,angle:angle,mirror:!mirror);
            Plan(a,b);
        }
        // Physically identical rectangles represented by swapped local axes and counts.
        var equivalent=Profile(500,400,5,4,Math.PI/2);
        var same=Plan(Profile(),equivalent);
        foreach(var p in same.Bars)Near((p.Lower-p.Upper).Length,0,"90-degree axis normalization");
        foreach(var p in Plan(Profile(),Profile(angle:Math.PI)).Bars)Near((p.Lower-p.Upper).Length,0,"180-degree family orientation");
        Plan(Profile(),Profile(angle:.1)); // physical rotation, not just axis relabelling
        var foot=Profile();foot.HasFoot=true;foot.FootDirection=V3AnchorDirection.PositiveY;
        Plan(foot,Profile(350,450));
        Reject(()=>V3ColumnPairing.Create(Profile(),Profile(cx:5),2,6),"Khác số thanh");
        Reject(()=>V3ColumnPairing.Create(Profile(cx:4,cy:5),Profile(cx:3,cy:6),2,6),"các cạnh");
        Reject(()=>V3ColumnPairing.Create(Profile(width:0),Profile(),2,6),"không hợp lệ");
        var bad=Profile();bad.AxisX=new V3Point2(2,0);
        Reject(()=>V3ColumnPairing.Create(bad,Profile(),2,6),"không hợp lệ");
        var far=Profile();far.Center=new V3Point2(5000,0);
        Reject(()=>V3ColumnPairing.Create(Profile(),far,2,6),"không có vị trí");
        Reject(()=>V3ColumnPairing.Create(Profile(400,400,4,4),Profile(400,400,4,4,Math.PI/4),2,6),"tương đương");
        var rows=V3ColumnPairing.Compare(Profile(),shifted);
        Check(rows.Single(r=>r.Parameter=="Cover (mm)").Status=="Khác","Expose actual cover difference");
        Check(rows.Single(r=>r.Parameter=="Thép chủ (mm)").Status=="Giống","Do not invent a diameter difference");
        // Run production adapter and resolver rather than a fake SameStructure boolean.
        var lower=new V3SegmentWork {Config=new V3RebarConfig {Cx=4,Cy=5,MainDiameter=20,TieDiameter=8,Cover=30,StarterBars=true,StarterLength=1200},
            Geometry=new V3Geometry {Bottom=0,Top=V3Units.Ft(3000)},Segment=new ColumnSegment {LevelName="Trệt"}};
        var upper=new V3SegmentWork {Config=lower.Config.Copy(),Geometry=new V3Geometry {Bottom=V3Units.Ft(3400),Top=V3Units.Ft(6400)},
            Segment=new ColumnSegment {LevelName="Lầu 1"}};
        upper.Config.StarterBars=false;upper.Config.TieDiameter=10;upper.Config.Cover=40;
        upper.Geometry.Width=V3Units.Ft(550);upper.Geometry.Depth=V3Units.Ft(750);
        V3StarterConnection.Resolve(new List<V3SegmentWork>{lower,upper});
        Check(lower.ConnectionPlan!=null&&lower.ConnectionPlan.Bars.Count==14,"Real resolver accepts changed cover/ties/section");
        upper.Config.Cx=5;
        Reject(()=>V3StarterConnection.Resolve(new List<V3SegmentWork>{lower,upper}),"Dưới:");

        var reduced=V3ColumnPairing.Create(Profile(200,300,3,3),Profile(200,300,2,2),2,6);
        Check(reduced.Bars.Count==4&&reduced.TerminatedLower.Count==4,"8-to-4 must connect corners and terminate four lower bars");
        Check(reduced.Bars.All(p=>V3ColumnPairing.Points(reduced.Lower)[p.LowerIndex].Corner),"Retain corner pairs when interior bars stop");
        Check(reduced.Bars.Select(p=>p.LowerIndex).Concat(reduced.TerminatedLower).Distinct().Count()==8,"No lower bar lost or duplicated");
        Check(reduced.Bars.Select(p=>p.UpperIndex).Distinct().Count()==4,"Upper bars still have exactly one pair");
        foreach(double angle in new[]{0.0,.7,Math.PI/2})foreach(bool mirror in new[]{false,true})
        {
            var r=V3ColumnPairing.Create(Profile(400,500,3,3,angle,mirror),Profile(400,500,2,2,angle,!mirror),2,6);
            Check(r.Bars.Count==4&&r.TerminatedLower.Count==4,"Reduction works in rotated and mirrored frames");
        }
        Check(!V3ColumnTerminationRules.SoffitInRectangle(new[]{new[]{new V3Point3(10,10,100),new V3Point3(20,10,200),new V3Point3(10,20,200)}},1,1).HasValue,"Nearby beam with no footprint intersection must not set the cutoff");
        Near(V3ColumnTerminationRules.SoffitInRectangle(new[]{new[]{new V3Point3(-10,0,100),new V3Point3(10,0,300),new V3Point3(0,10,300)}},1,1).Value,190,"Use the beam underside inside the column footprint, not the full beam bounding-box bottom");
        Near(V3ColumnTerminationRules.EndMm(0,3000,30,20,null),2960,"Unpaired bars end below column top, not at starter extension");
        Near(V3ColumnTerminationRules.EndMm(0,3000,30,20,2600),2560,"Detected beam soffit limits termination with end cover");
        Near(V3ColumnTerminationRules.EndMm(0,3000,30,20,3400),2960,"Never extend terminated bars above the lower column");
        Reject(()=>V3ColumnTerminationRules.EndMm(0,3000,30,20,double.NaN),"đáy dầm");
        Reject(()=>V3ColumnTerminationRules.EndMm(0,3000,30,20,50),"Chiều cao");

        lower.Config.Cx=3;lower.Config.Cy=3;upper.Config.Cx=2;upper.Config.Cy=2;
        V3StarterConnection.Resolve(new List<V3SegmentWork>{lower,upper});
        Check(lower.BottleneckToUpper&&lower.ConnectionPlan.Bars.Count==4&&lower.ConnectionPlan.TerminatedLower.Count==4,
            "Production resolver accepts reduction and retains explicit unmatched-bar identities");
        Check(V3ColumnPairing.Create(Profile(),Profile(),2,6).TerminatedLower.Count==0,"Equal counts retain all continuing bars");

        var paths=new List<V3BarPath>();
        for(int i=0;i<4;i++)paths.Add(new V3BarPath {BarIndex=i,Continues=true,Points=new List<V3Point3> {
            new V3Point3(i*100,0,0),new V3Point3(i*100,0,2800),new V3Point3(i*100,22,3000),new V3Point3(i*100,22,4000)}});
        var groups=V3ProfileGeometry.Group(paths,true);
        var stopped=new V3BarPath {Continues=false,Points=paths[0].Points.ToList()};
        Check(V3ProfileGeometry.Group(new List<V3BarPath>{paths[0],stopped},true).Count==2,"Never group a terminated bar with a continued bar, even if shapes coincide");
        Check(groups.Count==1&&groups[0].Bars.Count==4,"Translate identical planar profiles into a Fixed Number set");
        Check(V3ProfileGeometry.Group(paths,false).Count==4,"Single mode preserves bar count");
        paths[2].Points[2]=new V3Point3(205,22,3000);paths[2].Points[3]=new V3Point3(205,22,4000);
        groups=V3ProfileGeometry.Group(paths,true);
        Check(groups.Sum(g=>g.Bars.Count)==4&&groups.Count>1,"Different shape splits the group without losing bars");
        var spatial=new V3BarPath {Points=new List<V3Point3> {
            new V3Point3(-200,0,-600),new V3Point3(0,0,-600),new V3Point3(0,0,2700),new V3Point3(25,40,3000),new V3Point3(25,40,4000)}};
        spatial.Inspect();Check(!spatial.Planar,"Combined L foot and diagonal neck is non-planar");
        Check(V3ProfileGeometry.Group(new List<V3BarPath>{spatial},true).Single().Bars.Count==1,"Non-planar bars stay single");
        var rounded=V3ProfileGeometry.Round(spatial.Points,90,.5);
        Check(rounded.Count(c=>c.Arc)==3,"Round the foot and both neck bends in 3D");
        for(int i=1;i<rounded.Count;i++)Near((rounded[i].Start-rounded[i-1].End).Length,0,"Continuous rounded path");
        foreach(var arc in rounded.Where(c=>c.Arc))
        {
            var ab=arc.Mid-arc.Start;var ac=arc.End-arc.Start;
            double a=(arc.End-arc.Mid).Length,b=ac.Length,c=ab.Length;
            double radius=a*b*c/(2*ab.Cross(ac).Length);
            Near(radius,90,"Actual fillet circumradius");
        }
        Reject(()=>V3ProfileGeometry.Round(new List<V3Point3>{new V3Point3(0,0,0),new V3Point3(0,0,5),new V3Point3(5,0,5)},90,.5),"quá ngắn");
        Near(V3ProfileGeometry.SegmentDistance(new V3Point3(-10,0,0),new V3Point3(10,0,0),
            new V3Point3(0,-10,0),new V3Point3(0,10,0)),0,"Crossing segments");
        Near(V3ProfileGeometry.SegmentDistance(new V3Point3(-10,0,0),new V3Point3(10,0,0),
            new V3Point3(0,-10,25),new V3Point3(0,10,25)),25,"Skew segments");
        Near(V3ProfileGeometry.SegmentDistance(new V3Point3(0,0,0),new V3Point3(0,0,0),
            new V3Point3(20,0,-10),new V3Point3(20,0,10)),20,"Degenerate segment");
        Console.WriteLine($"PASS: {checks} production pairing/profile regression checks");
    }
}
